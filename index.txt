function mutateDigit(s) {
  s = String(s);

  for (let i = 0; i < s.length; i++) if (/\d/.test(s[i])) idx.push(i);

}

function openTab(type, id) {
  const prev = state.tabs.find(t => t.key === state.active);
  if (prev && ['ticket', 'logs'].includes(prev.type) && state.attention >= 40 && Math.random() < .25) gaslightMutate();
  const key = type + ':' + (id || '');
  let t = state.tabs.find(x => x.key === key);
  if (!t) { t = { key, type, id }; state.tabs.push(t); }
  state.active = key;
  if (type === 'chat') {
    state.unreadChat = false;
    if (state.phantomPending) {
      state.phantomPending = false;
      setTimeout(() => {
        if (!state.over) { pushChat('k', rnd(PHANTOM_MSGS)); renderAll(); }
      }, 9000 + Math.random() * 4000);
    }
  }
  renderAll();
}

function archNode(path) {
  let n = ARCHIVE;
  for (let i = 1; i < path.length; i++) {
    const next = (n.children || []).find(c => c.name === path[i]);
    if (!next) return null;
    n = next;
  }
  return n;
}

let routineUsed = [];
function makeRoutine() {
  let tpl;
  do { tpl = rnd(RT); } while (routineUsed.includes(tpl.subject) && routineUsed.length < RT.length);
  routineUsed.push(tpl.subject);
  let id;
  do { id = String(1000 + Math.floor(Math.random() * 9000)); } while (state.tickets[id]);
  const user = randUser();
  const logs = [
    lg(driftTime(), 'AUTH', user, 'AUTH_' + (Math.random() < .5 ? 'FAIL' : 'OK'), 'reason=' + rnd(['bad_password', 'timeout', 'ok'])),
    lg(driftTime(), rnd(['NODE-09', 'NODE-03', 'ROUTE-09']), user, 'LATENCY', 'value=' + (80 + Math.floor(Math.random() * 400)) + 'ms')
  ];
  if (Math.random() < .15) logs.push(lg(driftTime(), 'CLOCK', '', 'TIME_SYNC', 'error offset=+00:0' + (1 + Math.floor(Math.random() * 8)), 'bad'));
  return mkTicket({
    id, user, subject: tpl.subject, body: tpl.body, prio: rnd(['low', 'low', 'mid']),
    logs, actions: tpl.act.map(([label, act]) => ({ label, act })), routine: true
  });
}

function checkEnd() {
  if (state.over || state.betweenShifts || state.windDown || state.endOffered) return;
  const k = getKey();
  if (COMPLETE[k] && COMPLETE[k]()) beginWindDown();
}
function beginWindDown() {
  state.windDown = true;
  $('#endShift').classList.add('on');
  toast('Очередь пуста. Ночь затихает…', 'warn');
  let t = 12; /* 12 секунд атмосферной тишины */
  state.windTimer = setInterval(() => {
    if (state.over) { clearInterval(state.windTimer); return; }
    t--;
    if (t === 8) { dispMsg(rnd(WIND_LINES)); }
    if (t === 5 && Math.random() < .6) triggerDeadAirEvent();
    if (t <= 0) {
      clearInterval(state.windTimer);
      state.windTimer = null;
      finishShiftAuto();
    }
  }, 1000);
}
function finishShiftAuto() {
  state.windDown = false;
  state.minutes = Math.max(state.minutes, Math.min(360, 300));
  $('#clock').textContent = fmt(state.minutes);
  pushChat('s', 'Рабочий цикл завершён. Оставшиеся часы прошли тихо.');
  renderAll();
  setTimeout(() => { if (!state.over && !state.betweenShifts) endShiftNow(); }, 1500);
}
function checkTimeline() {
  if (!state.pendingTickets || !state.pendingTickets.length) return;
  const toSpawn = state.pendingTickets.filter(t => t.spawnTime <= state.minutes);
  toSpawn.forEach(t => {
    spawnTicket(t);
    toast('Новый тикет: #' + t.id, 'warn');
    state.pendingTickets = state.pendingTickets.filter(pt => pt.id !== t.id);
  });
  if (toSpawn.length) renderAll();
}

function tprint(text, cls) {
  const out = $('#termout');
  const d = document.createElement('div');
  if (cls) d.className = cls;
  d.textContent = text;
  out.appendChild(d);
  out.scrollTop = out.scrollHeight;
}
function termKey(e) {
  if (e.key === 'Enter') {
    const inp = $('#terminput');
    const raw = inp.value;
    inp.value = '';
    if (raw.trim()) { termHist.push(raw); termIdx = termHist.length; }
    tprint('OP-884@argus:~$ ' + raw, 'cmd');
    termExec(raw);
  }
  if (e.key === 'ArrowUp') {
    if (termIdx > 0) { termIdx--; $('#terminput').value = termHist[termIdx] || ''; }
    e.preventDefault();
  }
  if (e.key === 'ArrowDown') {
    if (termIdx < termHist.length) { termIdx++; $('#terminput').value = termHist[termIdx] || ''; }
    e.preventDefault();
  }
}
function logLine(l) { return l.t + ' ' + l.node + ' ' + l.user + ' ' + l.event + ' ' + l.details; }
function termExec(raw) {
  const a = raw.trim().split(/\s+/).filter(Boolean);
  if (!a.length) return;
  const c = a[0].toLowerCase();
  const HELP = 'Доступные команды:\nhelp, clear, status, whoami, sudo, time, history, save\ntickets | ticket show <id>\nlogs | logs grep <текст> | logs filter user=<x> | logs filter node=<x> | logs compare\nnodes list | node info <NODE-XX> | ping <NODE-XX>\ntrace hidden_route [--force]\nls | cat <путь> | mv <файл> | decode <файл|строка>\nunlock archive <код> | unlock lost_sessions <код>\ncp <путь>  (копировать улику)\nsec status | contact k.lanskaya | override reserve_channel\nФинал: isolate NODE-17 | upload evidence --external |\nrescue session 7741-KL | replace session OP-884 | delete all |\nupload evidence --full (секретно)';
  switch (c) {
    case 'help': tprint(HELP); break;
    case 'clear': $('#termout').innerHTML = ''; break;
    case 'status': tprint('Смена: ' + state.shift + (state.side ? ' (наряд ' + state.side + ')' : '') + '. Аудит: ' + state.attention + '%. Доверие: ' + state.trust + '. Связь: ' + state.link + '. Аптайм: ' + state.stability + '%.'); break;
    case 'whoami':
      if (state.attention >= 80 && Math.random() < .6) { setTimeout(() => tprint('7741-KL', 'err'), 700); }
      else tprint('OP-884');
      break;
    case 'sudo':
      if (state.attention >= 75) tprint('Доступ предоставлен.\nСеанс 7741-KL активен.\nВы уверены, что вы — это вы?', 'err');
      else tprint('OP-884 не входит в список доверенных операторов.');
      break;
    case 'time': tprint(fmt(state.minutes)); break;
    case 'save': saveGame(true); break;
    case 'history': {
      let h = termHist.slice();
      if (state.attention >= 60) {
        const fakeCmds = ['rm -rf /kira_archive', 'connect NODE-17', 'help me', 'whoami', 'sudo keep_me', 'I see you'];
        h.splice(Math.floor(Math.random() * h.length), 0, rnd(fakeCmds));
        if (Math.random() < .5) h.push(rnd(fakeCmds));
      }
      tprint(h.map((x, i) => (i + 1) + '  ' + x).join('\n') || 'История пуста.');
      break;
    }
    case 'tickets':
      tprint(Object.values(state.tickets).map(t => '#' + t.id + ' [' + t.status.toUpperCase() + '] ' + t.user + ': ' + t.subject).join('\n') || 'Очередь пуста.'); break;
    case 'ticket':
      if (a[1] === 'show' && state.tickets[a[2]]) { openTab('ticket', a[2]); tprint('Открываю тикет #' + a[2]); }
      else tprint('Использование: ticket show <id>', 'err'); break;
    case 'logs': {
      let rows = state.logs.slice();
      if (a[1] === 'grep' && a[2]) {
        const q = a.slice(2).join(' ').toLowerCase();
        rows = rows.filter(l => logLine(l).toLowerCase().includes(q));
        rows.forEach(l => {
          if (l.hidden) {
            state.flags.hiddenFound = state.flags.hiddenFound || {};
            state.flags.hiddenFound[l.event] = 1;
            if (Object.keys(state.flags.hiddenFound).length >= 3) ach('archivist');
          }
        });
      }
      else if (a[1] === 'filter' && a[2]) {
        const [k, v] = a[2].split('=');
        rows = rows.filter(l => (k === 'user' ? l.user : l.node) === v);
      }
      else if (a[1] === 'compare') {
        rows = rows.filter(l => /offset|drift/i.test(l.details));
        tprint(rows.length ? ('Расхождения:\n' + rows.map(logLine).join('\n')) : 'Расхождений не найдено.');
        if (rows.length) fact('f_time_letter');
        break;
      }
      else rows = rows.filter(l => !l.hidden || state.flags.advanced_search).slice(-12);
      tprint(rows.map(logLine).join('\n') || 'Ничего не найдено.');
      if (a[1] === 'grep' && a[2]) attn(3);
      break;
    }
    case 'nodes': tprint(NODES.map(nodeStateLine).join('\n')); break;
    case 'node':
      if (a[1] === 'info') {
        const n = NODES.find(x => x.id === (a[2] || '').toUpperCase());
        tprint(n ? nodeStateLine(n) : 'Узел не найден.', n ? '' : 'err');
      }
      else tprint('Использование: node info <NODE-XX>', 'err'); break;
    case 'ping': {
      const id = (a[1] || '').toUpperCase();
      if (id === 'NODE-17') {
        if (!state.node17Visible) { tprint('Узел не найден в активной таблице.', 'err'); break; }
        tprint('PING NODE-17: Request timeout...');
        setTimeout(() => { tprint('Unexpected reply from NODE-17.', 'err'); attn(3); sDanger(); }, 2400);
      } else if (NODES.find(n => n.id === id)) tprint(id + ': alive, 12 ms');
      else tprint('Неизвестный узел.', 'err');
      break;
    }
    case 'trace': {
      if (a[1] !== 'hidden_route') { tprint('Использование: trace hidden_route [--force]', 'err'); break; }
      if (state.shift < 4) { tprint('Маршрут неизвестен.', 'err'); break; }
      if (a[2] !== '--force') { tprint('Маршрут частично скрыт.\nТребуется флаг принудительной трассировки:\ntrace hidden_route --force'); break; }
      attn(10); trustUp(-3);
      advanceTime(25);
      tprint('01 CORE-GATE       12 ms\n02 ROUTE-09        18 ms\n03 ARCHIVE-04      41 ms\n04 NODE-17         response_without_source');
      state.node17Visible = true;
      state.flags.traced = true;
      addLog(lg(nowT(), 'TRACE', '', 'FORCED', 'dst=NODE-17 result=exposed', 'bad'));
      fact('f_node17_dead');
      toast('Узел NODE-17 появился на карте сети.', 'warn');
      renderAll();
      break;
    }
    case 'ls': {
      const node = archNode(state.archivePath);
      tprint((node.children || []).map(ch => (ch.dir ? '[DIR]  ' : '[FILE] ') + ch.name + ((ch.locked && ch.locked()) ? '  (locked)' : '')).join('\n'));
      break;
    }
    case 'mv': {
      const src = a[1] || '';
      if (src.includes('message_draft_01.dat')) {
        state.datRenamed = true;
        tprint('message_draft_01.dat -> message_draft_01.txt\nФайл переименован. Теперь его можно прочитать: cat /archive/kira_archive/message_draft_01.txt');
      } else tprint('Файл не найден.', 'err');
      break;
    }
    case 'decode': {
      let arg = a.slice(1).join(' ');
      if (arg.includes('message_draft_01')) {
        state.datRenamed = true;
        const node = (ARCHIVE.children.find(x => x.name === 'kira_archive').children || []).find(x => x.name === 'message_draft_01.dat');
        if (node) { if (node.onRead && !node.read) { node.read = true; node.onRead(); state.filesRead++; } tprint(node.content); }
        afterAct();
        break;
      }
      if (a[1] && a[1].startsWith('b64:')) arg = a.slice(1).join(' ').slice(4);
      const dec = decB64(arg.replace(/^b64:/, ''));
      if (dec) { tprint(dec); fact('f_b64'); ach('decoder'); linkUp(3); afterAct(); }
      else tprint('Не удаётся декодировать.', 'err');
      break;
    }
    case 'cat':
    case 'cp': {
      const path = (a.slice(1).join(' ') || '').replace(/^\/+/, '').split('/').filter(Boolean);
      if (path[0] !== 'archive') { tprint('Путь должен начинаться с /archive/', 'err'); break; }
      let node = ARCHIVE, ok = true;
      for (let i = 1; i < path.length; i++) {
        let nx = (node.children || []).find(ch => ch.name === path[i]);
        if (!nx && state.datRenamed && path[i] === 'message_draft_01.txt')
          nx = (node.children || []).find(ch => ch.name === 'message_draft_01.dat');
        if (!nx) { ok = false; break; }
        if (nx.locked && nx.locked()) { tprint('Доступ запрещён: /' + path.slice(0, i + 1).join('/') + '/', 'err'); ok = false; break; }
        node = nx;
      }
      if (!ok) break;
      if (!node.content && node.dir) { tprint('Это каталог.', 'err'); break; }
      if (node.isDat && !state.datRenamed) {
        tprint('50 4B 03 04 14 00 00 00 08 00 7A 4B\n[бинарные данные. текстовый слой не распознан]\nподсказка: decode ' + node.name + '  или  mv ' + node.name + ' ' + node.name.replace('.dat', '.txt'), 'err');
        break;
      }
      if (c === 'cat') { if (node.onRead && !node.read) { node.read = true; node.onRead(); state.filesRead++; afterAct(); } tprint(node.content); }
      else { if (node.ev) { addEvidence(node.ev); attn(5); afterAct(); } else tprint('Файл не является уликой.', 'err'); }
      break;
    }
    case 'unlock': {
      const code = (a[2] || '').toLowerCase();
      if (a[1] === 'archive' && code === 'chayka') { state.kiraUnlocked = true; tprint('Доступ к /kira_archive/ открыт.'); fact('f_kira_archive_open'); afterAct(); }
      else if (a[1] === 'lost_sessions' && code === 'node17_4471') { state.lostUnlocked = true; state.flags.opened_lost_sessions = true; fact('f_lost_sessions'); tprint('Доступ к /lost_sessions/ открыт.'); attn(5); afterAct(); }
      else tprint('Неверная команда или код.', 'err');
      break;
    }
    case 'sec': tprint('Уровень аудита: ' + state.attention + '%'); break;
    case 'override':
      if (a[1] === 'reserve_channel') {
        if (state.flags.reserve && !state.flags.reserveUsed) useReserve();
        else tprint('Резервный канал недоступен.', 'err');
      } else tprint('Использование: override reserve_channel', 'err');
      break;
    case 'contact':
      if (a[1] === 'k.lanskaya') {
        if (state.node17Connected) {
          tprint('CHANNEL OPEN...\n> Не пиши прямо. Он читает открытые строки.\n> Используй архив.');
          state.flags.contacted_kira = true;
          state.contactsKira++;
          linkUp(8);
          attn(10);
          /* Кира отвечает файлом в архиве */
          const ka = ARCHIVE.children.find(x => x.name === 'kira_archive');
          const n = Math.min(state.contactsKira, 3);
          const name = 'reply_0' + n + '.txt';
          if (ka && !ka.children.some(ch => ch.name === name)) {
            ka.children.push({
              name, content: KIRA_REPLIES[n - 1],
              onRead: () => { linkUp(4); state.flags['read_reply_' + n] = true; }
            });
            toast('В /kira_archive/ появился новый файл: ' + name, 'warn');
          }
          if (state.contactsKira >= 3) ach('ghost_talker');
          afterAct();
        }
        else tprint('Нет ответа.', 'err');
      } else tprint('Использование: contact k.lanskaya', 'err');
      break;
    case 'isolate': finalFromTerm('isolate'); break;
    case 'upload': finalFromTerm(a.includes('--full') ? 'full' : 'upload'); break;
    case 'rescue': finalFromTerm('rescue'); break;
    case 'replace': finalFromTerm('replace'); break;
    case 'delete': if (a[1] === 'all') finalFromTerm('delete'); else tprint('Использование: delete all', 'err'); break;
    default: tprint('Команда не найдена: ' + c + '. См. help.', 'err');
  }
  advanceTime(2); /* любая команда терминала двигает время */
}

}

function gaslightMutate() {
  const opts = [];
  if (state.logs.length) opts.push('log');
  const openT = Object.values(state.tickets).filter(t => t.status === 'open');
  if (openT.length) opts.push('ticket');
  if (!opts.length) return;
  if (rnd(opts) === 'log') {
    const l = rnd(state.logs);
    if (!l.isMutated && !l.originalDetails) {
      l.originalDetails = l.details;
      l.originalT = l.t;
      l.details = mutateDigit(l.details);
      if (Math.random() < .5) l.t = mutateDigit(l.t);
      l.isMutated = true;
    }
  } else {
    const t = rnd(openT);
    if (!t.isMutated && !t.originalBody) {
      t.originalBody = t.body;
      t.body = mutateDigit(t.body);
      t.isMutated = true;
    }
  }
}

function collectRead(node, arr) {
  (node.children || []).forEach(c => {
    if (c.read) arr.push(c.name);
    if (c.dir) collectRead(c, arr);
  });
}
function markRead(node, names) {
  (node.children || []).forEach(c => {
    if (names.includes(c.name)) c.read = true;
    if (c.dir) markRead(c, names);
  });
}

    lastActTime: Date.now(), runAch: [], quietOpen: {}, ghostSug: null,
    betweenShifts: false, windDown: false, windTimer: null
  };

function loadSave() {
  try {
    const d = JSON.parse(localStorage.getItem(SAVE_KEY));
    if (!d) return false;
    const keep = state.muted;
    Object.assign(state, d.state);
    state.over = false;
    state.muted = keep;
    state.windDown = false;
    state.betweenShifts = false;
    patchState();
    markRead(ARCHIVE, d.read || []);
    state.tabs = state.tabs || [];
    if (!state.tabs.length) state.tabs = [{ key: 'tickets:', type: 'tickets' }];
    if (!state.tabs.find(t => t.key === state.active)) state.active = state.tabs[0].key;
    state.started = true;
    renderAll();
    startAmbient();
    updateAmbient();
    return true;
  } catch (e) { return false; }
}

/* Мета-тик: умный спавн рутины + финальный таймаут */
function metaTick() {
  if (state.over || !state.started || state.betweenShifts || state.windDown) return;
  /* 1) умная рутина: заполняем пустоту, когда игроку нечего делать */
  const k = getKey();
  const openCloseable = Object.values(state.tickets).filter(t => t.status === 'open' && t.closeable).length;
  const done = COMPLETE[k] && COMPLETE[k]();
  if (openCloseable === 0 && !state.pendingTickets.length && !done && state.routineSpawned < routineCap(k)) {
    spawnRoutineSmart(1);
    renderAll();
  }
  /* 2) финальный таймаут: в смене 3 минуты бездействия — система решает сама */
  if (state.shift === 6 && !state.side) {
    const idle = Date.now() - state.lastActTime;
    if (!state.finalWarned && idle > 120000) {
      state.finalWarned = true;
      toast('СИСТЕМА: решение не получено. Осталось 60 секунд.', 'bad');
      sDanger();
      renderAll();
    }
    if (idle > 180000) ending('END_TIMEOUT');
  }
}
